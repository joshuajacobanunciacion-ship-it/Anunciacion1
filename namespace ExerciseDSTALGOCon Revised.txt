namespace ExerciseDSTALGOCon
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void button1_Click(object sender, EventArgs e)
        {
            
            if (double.TryParse(textBox1.Text, out double number))
            {
                string sign;
                if (number > 0)
                    sign = "positive";
                else if (number < 0)
                    sign = "negative";
                else
                    sign = "zero";

                label2.Text = ("You entered" + number + " It is " + sign );
            }
            else
            {
                label2.Text = "Please enter a valid number.";
            }
        }
    }
}
