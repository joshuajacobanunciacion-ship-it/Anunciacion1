namespace ExerciseDSTALGOCon
{
    public partial class Form1 : Form
    {
        public Form1()
        {
            InitializeComponent();
        }

        private void button1_Click(object sender, EventArgs e)
        {
            // Try to parse as a floating-point number so decimals and negatives are supported
            if (double.TryParse(textBox1.Text, out double number))
            {
                label2.Text = number.ToString();
            }
            else
            {
                // If parsing fails, just display the raw input
                label2.Text = textBox1.Text;
            }
        }
    }
}
